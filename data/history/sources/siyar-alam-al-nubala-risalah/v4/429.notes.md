= ورواه البخاري ٣ / ١٦٣، ومسلم (٩٥١) ، وأبو داود (٣٢٠٤) ، والطيالسي (٢٣٠٠) ، وابن ماجه (١٥٣٤) والنسائي ٤ / ٧٠، والترمذي (١٠٢٢) من حديث أبي هريرة.وأخرجه مسلم (٩٥٣) ، والنسائي ٤ / ٧٠، وابن ماجه (١٥٣٥) ، والطيالسي (٧٤٩) ، وأحمد ٤ / ٤٣١، ٤٣٣، والترمذي (١٠٣٩) من حديث عمران بن حصين.ورواه الطيالسي (١٠٦٨) ، وابن ماجه (١٥٣٧) ، وأحمد ٤ / ٧ عن حذيفة بن أسيد.ورواه أحمد ٤ / ٦٤، وابن ماجه (١٥٣٦) ، وأحمد ٥ / ٣٧٦ من حديث مجمع بن حارثة الأنصاري.وأخرجه ابن ماجه (١٥٣٨) من حديث عبد الله بن عمر.وأخرجه أحمد ٤ / ٢٦٠، ٢٦٣ من حديث جرير بن عبد الله.

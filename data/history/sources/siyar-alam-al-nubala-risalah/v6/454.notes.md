(١) العلاء بن يزيد ترجمه ابن أبي حاتم ٦ / ٣٦٢ فلم يذكر فيه جرحا ولا تعديلا، وباقي رجاله ثقات.وهو عند ابن عساكر ١٤ / ٦٧ آ.(٢) رجاله ثقات، وسنده قوي.وهو في " المسند " ٤ / ١٠٥ و٥ / ٢٩٠، وابن سعد ٧ / ٤٢٩، وابن عساكر ١٤ / ٦٧ آ.(٣) إسناده صحيح، وأخرجه أحمد ٥ / ١٤٥ من طريق يونس وعفان، بهذا الإسناد، وأخرجه أحمد ٥ / ١٦٥ و١٧٧، وأبو داود (٢٩٦٢) ، وابن ماجه (١٠٨) من طريق ابن إسحاق، عن مكحول، عن غضيف، عن أبي ذر، وصححه الحاكم ٣ / ٨٦، ٨٧ ووافقه الذهبي.
